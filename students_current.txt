const FEMALE_SUFFIXES = [
  'ة',
  'اء',
  'ى',
];

function normalizeArabicName(name: string): string {
  return name
    .trim()
    .toLowerCase()
    .replace(/[إأآا]/g, 'ا')
    .replace(/ى/g, 'ي')
    .replace(/ة/g, 'ه')
    .replace(/\s+/g, ' ');
}

function detectGenderFromArabicName(name: string): Gender {
  const normalized = normalizeArabicName(name);
  const firstName = normalized.split(' ')[0];

  if (!firstName) {
    return 'Male';
  }

  const normalizedFemaleNames = new Set(
    Array.from(FEMALE_NAMES).map(normalizeArabicName)
  );

  if (normalizedFemaleNames.has(firstName)) {
    return 'Female';
  }

  if (FEMALE_SUFFIXES.some((suffix) => firstName.endsWith(suffix))) {
    return 'Female';
  }

  return 'Male';
}

     OPEN ADD
  ===================================================== */

      ...emptyStudent(),

     OPEN EDIT
  ===================================================== */

     ARABIC NAME CHANGE

      ...previous,
      nameAr: value,

     SAVE
  ===================================================== */

/* =====================================================
   REUSABLE COMPONENTS
===================================================== */

function SectionTitle({
  title,
}: {
  title: string;
}) {
  return (
    <div className="mb-4 mt-8 first:mt-0">
      <h3 className="border-b border-gray-200 pb-2 text-sm font-bold uppercase tracking-wide text-gray-700">
        {title}
      </h3>
    </div>
  );
}

function Input({
  label,
  value,
  onChange,
  type = 'text',
  dir,
  placeholder,
}: {
  label: string;
  value: string;
  onChange: (value: string) => void;
  type?: string;
  dir?: 'ltr' | 'rtl';
  placeholder?: string;
}) {
  return (
    <div>
      <label className="mb-1.5 block text-sm font-medium text-gray-700">
        {label}
      </label>

      <input
        type={type}
        value={value}
        dir={dir}
        placeholder={placeholder}
        onChange={(e) => onChange(e.target.value)}
        className="w-full rounded-lg border border-gray-300 bg-gray-50 px-3 py-2.5 text-sm outline-none transition focus:border-gray-500 focus:bg-white"
      />
    </div>
  );
}

function TextArea({
  label,
  value,
  onChange,
}: {
  label: string;
  value: string;
  onChange: (value: string) => void;
}) {
  return (
    <div className="mt-4">
      <label className="mb-1.5 block text-sm font-medium text-gray-700">
        {label}
      </label>

      <textarea
        value={value}
        onChange={(e) => onChange(e.target.value)}
        rows={3}
        className="w-full resize-none rounded-lg border border-gray-300 bg-gray-50 px-3 py-2.5 text-sm outline-none transition focus:border-gray-500 focus:bg-white"
      />
    </div>
  );
}

function Select({
  label,
  value,
  onChange,
  options,
}: {
  label: string;
  value: string;
  onChange: (value: string) => void;
  options: {
    value: string;
    label: string;
  }[];
}) {
  return (
    <div>
      <label className="mb-1.5 block text-sm font-medium text-gray-700">
        {label}
      </label>

      <select
        value={value}
        onChange={(e) => onChange(e.target.value)}
        className="w-full rounded-lg border border-gray-300 bg-gray-50 px-3 py-2.5 text-sm outline-none focus:border-gray-500"
      >
        {options.map((option) => (
          <option
            key={option.value}
            value={option.value}
          >
            {option.label}
          </option>
        ))}
      </select>
    </div>
  );
}

function SkillSelect({
  label,
  value,
  onChange,
}: {
  label: string;
  value?: SkillLevel;
  onChange: (value: SkillLevel) => void;
}) {
  return (
    <Select
      label={label}
      value={value || 'Not assessed'}
      onChange={(value) =>
        onChange(value as SkillLevel)
      }
      options={options([
        'Strong',
        'Good',
        'Developing',
        'Needs Support',
        'Not assessed',
      ])}
    />
  );
}

function options(
  values: string[]
) {
  return values.map((value) => ({
    value,
    label: value,
  }));
}
